import { createClient } from "https://esm.sh/@supabase/supabase-js@2"; import { SignJWT, importPKCS8 } from "https://esm.sh/jose@5"; const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", "Access-Control-Allow-Methods": "POST, OPTIONS" }; Deno.serve(async (req) => { console.log("JAAS FUNCTION START"); console.log("METHOD:", req.method); if (req.method === "OPTIONS") { return new Response("ok", { headers: corsHeaders }); } try { const supabaseUrl = Deno.env.get("SUPABASE_URL"); const supabaseAnonKey = Deno.env.get("SUPABASE_ANON_KEY"); const appId = Deno.env.get("JAAS_APP_ID"); const kid = Deno.env.get("JAAS_KID"); const privateKey = Deno.env.get("JAAS_PRIVATE_KEY"); if (!supabaseUrl || !supabaseAnonKey) { throw new Error( "Missing Supabase environment variables" ); } if (!appId || !kid || !privateKey) { throw new Error( "Missing JAAS secrets" ); } const authHeader = req.headers.get("Authorization"); if (!authHeader) { return new Response( JSON.stringify({ error: "Missing Authorization header" }), { status: 401, headers: { ...corsHeaders, "Content-Type": "application/json" } } ); } const supabase = createClient( supabaseUrl, supabaseAnonKey, { global: { headers: { Authorization: authHeader } } } ); const { data: { user }, error: authError } = await supabase.auth.getUser(); if (authError) { console.error( "AUTH ERROR:", authError ); return new Response( JSON.stringify({ error: "Authentication failed", details: authError.message }), { status: 401, headers: { ...corsHeaders, "Content-Type": "application/json" } } ); } if (!user) { return new Response( JSON.stringify({ error: "No authenticated user" }), { status: 401, headers: { ...corsHeaders, "Content-Type": "application/json" } } ); } let body: any; try { body = await req.json(); } catch (e) { throw new Error( "Invalid JSON body" ); } const room = body && body.room; const callId = body && body.call_id ? String(body.call_id) : ""; if ( typeof room !== "string" || !/^[A-Za-z0-9_-]{1,100}$/.test(room) ) { throw new Error( "Invalid room" ); } console.log( "AUTH USER:", user.id ); console.log( "ROOM:", room ); console.log( "CALL ID:", callId ); const { data: profile, error: profileError } = await supabase .from("users") .select( "id,email,name,avatar,role" ) .eq( "id", user.id ) .single(); if (profileError) { console.error( "PROFILE ERROR:", profileError ); throw new Error( "Could not load user profile: " + profileError.message ); } if (!profile) { throw new Error( "User profile not found" ); } /* * The authenticated Supabase * user ID is the JaaS participant * identity. * * This MUST be different for * different LiveMuse accounts. */ const participantId = String(user.id); const participantName = profile.name && String(profile.name).trim().length > 0 ? String(profile.name).trim() : "LiveMuse User"; const participantEmail = profile.email ? String(profile.email) : ""; const participantAvatar = profile.avatar ? String(profile.avatar) : ""; const moderator = profile.role === "creator"; console.log( "JAAS PARTICIPANT ID:", participantId ); console.log( "JAAS PARTICIPANT NAME:", participantName ); console.log( "JAAS MODERATOR:", moderator ); const signingKey = await importPKCS8( privateKey, "RS256" ); const now = Math.floor( Date.now() / 1000 ); /* * JaaS JWT. * * room is intentionally the * same for both participants. * * context.user.id is what * identifies each participant. */ const token = await new SignJWT({ room: room, context: { user: { id: participantId, name: participantName, avatar: participantAvatar, email: participantEmail, moderator: moderator ? "true" : "false" }, features: { livestreaming: false, recording: false, transcription: false, "outbound-call": false } } }) .setProtectedHeader({ alg: "RS256", typ: "JWT", kid: kid }) .setAudience("jitsi") .setIssuer("chat") .setSubject(appId) .setNotBefore( now - 10 ) .setExpirationTime( now + 3600 ) .sign( signingKey ); console.log( "JAAS JWT CREATED" ); return new Response( JSON.stringify({ token: token, room_name: appId + "/" + room, participant_id: participantId, participant_name: participantName, moderator: moderator }), { status: 200, headers: { ...corsHeaders, "Content-Type": "application/json" } } ); } catch (error) { console.error( "JAAS FUNCTION ERROR:", error ); return new Response( JSON.stringify({ error: error instanceof Error ? error.message : String(error) }), { status: 500, headers: { ...corsHeaders, "Content-Type": "application/json" } } ); } });
